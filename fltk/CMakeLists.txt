cmake_minimum_required(VERSION 3.21)
project(penguin-mail-fltk VERSION 1.0.10 LANGUAGES C CXX)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
find_program(PENGUIN_STRIP_EXECUTABLE NAMES strip llvm-strip REQUIRED)
option(PENGUIN_ENABLE_MISFIN "Build optional Misfin(C) messaging" OFF)
set(ENGINE_FEATURE_ARGS)
if(PENGUIN_ENABLE_MISFIN)
  set(ENGINE_FEATURE_ARGS --features misfin)
endif()
set(ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(ENGINE_CARGO_ARGS)
  set(ENGINE_CARGO_DIRECTORY debug)
else()
  set(ENGINE_CARGO_ARGS --release)
  set(ENGINE_CARGO_DIRECTORY release)
endif()
# Re-run Cargo when compiled-in OAuth configuration changes. Store only a digest.
string(SHA256 PENGUIN_OAUTH_BUILD_KEY "$ENV{PENGUIN_MAIL_GOOGLE_CLIENT_ID}|$ENV{PENGUIN_MAIL_GOOGLE_CLIENT_SECRET}|$ENV{PENGUIN_MAIL_MICROSOFT_CLIENT_ID}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/oauth-client-build.stamp" CONTENT "@PENGUIN_OAUTH_BUILD_KEY@\nmisfin=@PENGUIN_ENABLE_MISFIN@\n" @ONLY)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${ENGINE_DIR}/*/src/*.rs" "${ENGINE_DIR}/*/Cargo.toml" "${CMAKE_CURRENT_SOURCE_DIR}/engine/src/*.rs" "${CMAKE_CURRENT_SOURCE_DIR}/engine/Cargo.toml")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/penguin-mail-engine"
  COMMAND "${CARGO_EXECUTABLE}" build --manifest-path "${ENGINE_DIR}/Cargo.toml" --locked -p penguin-mail-engine ${ENGINE_CARGO_ARGS} ${ENGINE_FEATURE_ARGS}
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${ENGINE_DIR}/target/${ENGINE_CARGO_DIRECTORY}/penguin-mail-engine" "${CMAKE_CURRENT_BINARY_DIR}/penguin-mail-engine"
  COMMAND "${PENGUIN_STRIP_EXECUTABLE}" --strip-debug "${CMAKE_CURRENT_BINARY_DIR}/penguin-mail-engine"
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/oauth-client-build.stamp" ${ENGINE_SOURCES} "${ENGINE_DIR}/Cargo.lock" "${ENGINE_DIR}/Cargo.toml"
  USES_TERMINAL)
add_custom_target(penguin-mail-core ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/penguin-mail-engine")
add_executable(penguin-mail native/main.cpp)
add_dependencies(penguin-mail penguin-mail-core)
if(PENGUIN_ENABLE_MISFIN)
  target_compile_definitions(penguin-mail PRIVATE PENGUIN_ENABLE_MISFIN=1)
endif()
target_compile_features(penguin-mail PRIVATE cxx_std_17)
target_compile_options(penguin-mail PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(penguin-mail PRIVATE native)
target_link_libraries(penguin-mail PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG)
set_target_properties(penguin-mail PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>;$<TARGET_FILE_DIR:fltk::gl-shared>")
include(native/desktop-menu.cmake)
de_xr_desktop_menu(penguin-mail)
include(GNUInstallDirs)
install(TARGETS penguin-mail RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/penguin-mail-engine" DESTINATION ${CMAKE_INSTALL_BINDIR})
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/io.github.c9dev.PenguinMail.desktop" PENGUIN_DESKTOP_ENTRY)
if(PENGUIN_ENABLE_MISFIN)
  string(REPLACE "x-scheme-handler/mailto;" "x-scheme-handler/mailto;x-scheme-handler/misfin;" PENGUIN_DESKTOP_ENTRY "${PENGUIN_DESKTOP_ENTRY}")
endif()
# Retain DomeD realm routing for links and desktop actions on every install.
find_program(PENGUIN_DOME_LINK_EXECUTABLE NAMES dome-link)
set(PENGUIN_DESKTOP_COMMAND "\"${CMAKE_INSTALL_FULL_BINDIR}/penguin-mail\"")
if(PENGUIN_DOME_LINK_EXECUTABLE)
  set(PENGUIN_DESKTOP_COMMAND "\"${PENGUIN_DOME_LINK_EXECUTABLE}\" --app-id io.github.c9dev.PenguinMail -- ${PENGUIN_DESKTOP_COMMAND}")
endif()
string(REPLACE "Exec=penguin-mail" "Exec=${PENGUIN_DESKTOP_COMMAND}" PENGUIN_DESKTOP_ENTRY "${PENGUIN_DESKTOP_ENTRY}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/io.github.c9dev.PenguinMail.desktop" "${PENGUIN_DESKTOP_ENTRY}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/io.github.c9dev.PenguinMail.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(DIRECTORY "${ENGINE_DIR}/app/data/icons/" DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor)

install(DIRECTORY "${ENGINE_DIR}/target/locale/" DESTINATION ${CMAKE_INSTALL_DATADIR}/locale OPTIONAL)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(penguin-tray-test tests/tray.cpp)
  target_compile_features(penguin-tray-test PRIVATE cxx_std_17)
  target_include_directories(penguin-tray-test PRIVATE native)
  target_link_libraries(penguin-tray-test PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG)
  de_xr_desktop_menu(penguin-tray-test)
  add_test(NAME login-and-tray COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tray.py $<TARGET_FILE:penguin-tray-test>)
endif()
